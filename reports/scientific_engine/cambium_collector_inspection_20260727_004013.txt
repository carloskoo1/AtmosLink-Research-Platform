   138	def run_ssh(cmd):
   139	    command = [
   140	        "sshpass", "-f", PASSFILE,
   141	        "ssh",
   142	        "-o", "StrictHostKeyChecking=no",
   143	        "-o", "UserKnownHostsFile=/dev/null",
   144	        "-o", f"ConnectTimeout={SSH_TIMEOUT}",
   145	        f"{SSH_USER}@{AP_IP}",
   146	        cmd,
   147	    ]
   148	
   149	    result = subprocess.run(
   150	        command,
   151	        capture_output=True,
   152	        text=True,
   153	        timeout=SSH_TIMEOUT + 5
   154	    )
   155	
   156	    if result.returncode != 0:
   157	        raise RuntimeError(result.stderr.strip())
   158	
   159	    return result.stdout

   162	def value_after_key(text, key):
   163	    for line in text.splitlines():
   164	        parts = line.strip().split()
   165	        if len(parts) >= 2 and parts[0] == key:
   166	            return parts[1]
   167	    return None
   168	
   169	
   170	def value_after_colon(text, key):
   171	    for line in text.splitlines():
   172	        if key in line and ":" in line:
   173	            return line.split(":", 1)[1].strip()
   174	    return None

   177	def classify_note(row):
   178	    note = "ok"
   179	
   180	    snr_dl = to_float(row.get("snr_dl"))
   181	    rssi_dl = to_float(row.get("sta_dl_rssi"))
   182	    mcs_dl = to_float(row.get("mcs_dl"))
   183	    dl_rate = to_float(row.get("dl_rate"))
   184	
   185	    if snr_dl is not None and snr_dl < 15:
   186	        note = "LOW_SNR"
   187	
   188	    if rssi_dl is not None and rssi_dl < -75:
   189	        note = "LOW_RSSI"
   190	
   191	    if mcs_dl is not None and mcs_dl < 3:
   192	        note = "LOW_MCS"
   193	
   194	    if dl_rate is not None and dl_rate < 20:
   195	        note = "LOW_RATE"
   196	
   197	    return note

   298	def run_daemon():
   299	    init_db()
   300	
   301	    print("AtmosLink RadioLink Collector iniciado")
   302	    print(f"Config: {CONFIG.get('_config_file', 'unknown')}")
   303	    print(f"Station: {STATION_ID} | {STATION_NAME}")
   304	    print(f"Role: {RADIO_ROLE} | Local role: {LOCAL_ROLE}")
   305	    print(f"AP: {AP_IP}")
   306	    print(f"SM: {SM_IP}")
   307	    print(f"Intervalo: {INTERVAL_SECONDS} s")
   308	
   309	    while True:
   310	        try:
   311	            collect_once()
   312	
   313	        except KeyboardInterrupt:
   314	            print("RadioLink Collector detenido por usuario")
   315	            break
   316	
   317	        except Exception as e:
   318	            err = str(e)
   319	            print(f"RADIO ERROR: {err}")
   320	            try:
   321	                save_status("RADIO_UNAVAILABLE", err)
   322	            except Exception as db_err:
   323	                print(f"ERROR GUARDANDO ESTADO RADIO: {db_err}")
   324	
   325	        time.sleep(INTERVAL_SECONDS)
   326	
   327	
   328	if __name__ == "__main__":
   329	    run_daemon()

    23	extract_first_value() {
    24	    local text="$1"
    25	    local key="$2"
    26	    echo "$text" | awk -v k="$key" '
    27	        index($0,k)>0 {
    28	            for(i=1;i<=NF;i++){
    29	                if($i==k && (i+1)<=NF){
    30	                    print $(i+1)
    31	                    exit
    32	                }
    33	            }
    34	        }'
    35	}
    36	
    37	normalize_value() {
    38	    local v="${1:-}"
    39	    v="$(echo "$v" | tr -d '\r' | xargs 2>/dev/null || true)"
    40	    echo "$v"
    41	}
    42	
    43	sanitize_csv_field() {
    44	    local v="${1:-}"
    45	    v="$(echo "$v" | tr -d '\r\n' | tr ',' ';')"
    46	    echo "$v"
    47	}
    48	
    49	while true; do
    50	    TS="$(date --iso-8601=seconds)"
    51	
    52	    RAW="$(run_show_sta || true)"
    53	
    54	    echo "===== $TS =====" >> "$RAW_FILE"
    55	    echo "$RAW" >> "$RAW_FILE"
    56	    echo >> "$RAW_FILE"
    57	
    58	    if [[ -z "$(echo "$RAW" | xargs 2>/dev/null || true)" ]]; then
    59	        printf "%s,%s,%s,,,,,,,,,,%s\n" \
    60	        "$TS" "AP" "$AP_IP" "empty_show_sta" >> "$CSV_FILE"
    61	        sleep "$INTERVAL_SECONDS"
    62	        continue
    63	    fi
    64	
    65	    PEER_IP="$(extract_first_value "$RAW" "connectedSTAIP")"
    66	    DL_MCS="$(extract_first_value "$RAW" "connectedSTADLMCS")"
    67	    UL_MCS="$(extract_first_value "$RAW" "connectedSTAULMCS")"
    68	    DL_SNR="$(extract_first_value "$RAW" "connectedSTADLSNR")"
    69	    UL_SNR="$(extract_first_value "$RAW" "connectedSTAULSNR")"
    70	    DL_RSSI="$(extract_first_value "$RAW" "connectedSTADLRSSI")"
    71	    UL_RSSI="$(extract_first_value "$RAW" "connectedSTAULRSSI")"
    72	    DL_RATE="$(extract_first_value "$RAW" "connectedSTADLRateMbps")"
    73	
    74	    PEER_IP="$(sanitize_csv_field "$(normalize_value "$PEER_IP")")"
    75	    DL_MCS="$(sanitize_csv_field "$(normalize_value "$DL_MCS")")"
    76	    UL_MCS="$(sanitize_csv_field "$(normalize_value "$UL_MCS")")"
    77	    DL_SNR="$(sanitize_csv_field "$(normalize_value "$DL_SNR")")"
    78	    UL_SNR="$(sanitize_csv_field "$(normalize_value "$UL_SNR")")"
    79	    DL_RSSI="$(sanitize_csv_field "$(normalize_value "$DL_RSSI")")"
    80	    UL_RSSI="$(sanitize_csv_field "$(normalize_value "$UL_RSSI")")"
    81	    DL_RATE="$(sanitize_csv_field "$(normalize_value "$DL_RATE")")"
    82	
    83	    NOTE="ok"
    84	    # ALERTAS
    85	if [[ ! -z "$DL_SNR" && "$DL_SNR" -lt 15 ]]; then
    86	    NOTE="LOW_SNR"
    87	fi
    88	
    89	if [[ ! -z "$DL_RSSI" && "$DL_RSSI" -lt -75 ]]; then
    90	    NOTE="LOW_RSSI"
    91	fi
    92	
    93	if [[ ! -z "$DL_MCS" && "$DL_MCS" -lt 3 ]]; then
    94	    NOTE="LOW_MCS"
    95	fi
    96	
    97	if [[ ! -z "$DL_RATE" && "$DL_RATE" -lt 20 ]]; then
    98	    NOTE="LOW_RATE"
    99	fi
   100	
   101	    printf "%s,%s,%s,%s,%s,%s,%s,%s,%s,%s,%s,%s\n" \
   102	    "$TS" "AP" "$AP_IP" "$PEER_IP" "$DL_MCS" "$UL_MCS" \
   103	    "$DL_SNR" "$UL_SNR" "$DL_RSSI" "$UL_RSSI" "$DL_RATE" "$NOTE" >> "$CSV_FILE"
   104	
   105	    sleep "$INTERVAL_SECONDS"
   106	done
